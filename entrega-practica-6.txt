--- app.js ---
const express = require('express')

const app = express()
const PORT = 3000

// Middleware para que Express pueda leer el cuerpo de las peticiones en formato JSON
app.use(express.json())

// "Base de datos" en memoria: los datos se pierden al reiniciar el servidor
let productos = [
    {
        id: 1,
        nombre: 'Laptop 15 pulgadas',
        descripcion: 'Laptop con 16 GB de RAM y SSD de 512 GB',
        precio: 15000
    },
    {
        id: 2,
        nombre: 'Mouse inalambrico',
        descripcion: 'Mouse optico inalambrico',
        precio: 450
    },
    {
        id: 3,
        nombre: 'Monitor 23.5 pulgadas full hd',
        descripcion: 'Monitor Full HD panel IPS',
        precio: 6500
    }
]

// ---------------------------------------------------------------
// Funciones auxiliares de validación
// ---------------------------------------------------------------

// Revisa que vengan los tres campos obligatorios en el cuerpo de la petición
function faltanCampos(datos) {
    const { nombre, descripcion, precio } = datos
    return (
        nombre === undefined || nombre === '' ||
        descripcion === undefined || descripcion === '' ||
        precio === undefined || precio === ''
    )
}

// Reto opcional: el precio debe ser un número mayor que cero
function precioValido(precio) {
    const precioNumerico = parseFloat(precio)
    return !isNaN(precioNumerico) && precioNumerico > 0
}

// ---------------------------------------------------------------
// GET /productos - Listar todos los productos
// ---------------------------------------------------------------
app.get('/productos', (req, res) => {
    res.json(productos)
})

// ---------------------------------------------------------------
// GET /productos/:id - Obtener un producto por id
// ---------------------------------------------------------------
app.get('/productos/:id', (req, res) => {
    const id = parseInt(req.params.id)
    const producto = productos.find(p => p.id === id)
    if (!producto) {
        return res.status(404).json({ mensaje: 'Producto no encontrado' })
    }
    res.json(producto)
})

// ---------------------------------------------------------------
// POST /productos - Crear nuevo producto
// ---------------------------------------------------------------
app.post('/productos', (req, res) => {
    const datos = req.body || {}

    // Validar que vengan nombre, descripcion y precio
    if (faltanCampos(datos)) {
        return res.status(400).json({
            mensaje: 'Los campos nombre, descripcion y precio son obligatorios'
        })
    }

    // Validar que el precio sea un número mayor que cero
    if (!precioValido(datos.precio)) {
        return res.status(400).json({
            mensaje: 'El precio debe ser un numero mayor que cero'
        })
    }

    // Generar un id único: el id más alto existente + 1 (o 1 si el arreglo está vacío)
    const nuevoId = productos.length > 0
        ? Math.max(...productos.map(p => p.id)) + 1
        : 1

    const nuevoProducto = {
        id: nuevoId,
        nombre: datos.nombre,
        descripcion: datos.descripcion,
        precio: parseFloat(datos.precio)
    }

    productos.push(nuevoProducto)

    // 201 Created con el producto creado
    res.status(201).json(nuevoProducto)
})

// ---------------------------------------------------------------
// PUT /productos/:id - Actualizar producto
// ---------------------------------------------------------------
app.put('/productos/:id', (req, res) => {
    const id = parseInt(req.params.id)
    const indice = productos.findIndex(p => p.id === id)

    // Si no existe el producto, responder 404
    if (indice === -1) {
        return res.status(404).json({ mensaje: 'Producto no encontrado' })
    }

    const datos = req.body || {}

    // Validar que vengan nombre, descripcion y precio
    if (faltanCampos(datos)) {
        return res.status(400).json({
            mensaje: 'Los campos nombre, descripcion y precio son obligatorios'
        })
    }

    // Validar que el precio sea un número mayor que cero
    if (!precioValido(datos.precio)) {
        return res.status(400).json({
            mensaje: 'El precio debe ser un numero mayor que cero'
        })
    }

    // Reemplazar los datos conservando el mismo id
    const productoActualizado = {
        id: id,
        nombre: datos.nombre,
        descripcion: datos.descripcion,
        precio: parseFloat(datos.precio)
    }

    productos[indice] = productoActualizado

    // 200 OK con el producto actualizado
    res.status(200).json(productoActualizado)
})

// ---------------------------------------------------------------
// DELETE /productos/:id - Eliminar producto
// ---------------------------------------------------------------
app.delete('/productos/:id', (req, res) => {
    const id = parseInt(req.params.id)
    const indice = productos.findIndex(p => p.id === id)

    // Si no existe el producto, responder 404
    if (indice === -1) {
        return res.status(404).json({ mensaje: 'Producto no encontrado' })
    }

    // splice devuelve un arreglo con los elementos eliminados
    const [productoEliminado] = productos.splice(indice, 1)

    res.status(200).json({
        mensaje: 'Producto eliminado',
        producto: productoEliminado
    })
})

// ---------------------------------------------------------------
// Iniciar el servidor
// ---------------------------------------------------------------
app.listen(PORT, () => {
    console.log(`Servidor escuchando en http://localhost:${PORT}`)
})
